\begin{esercizio}
\[
    \left\{[(24^2)^3]^3:(12^4)^4:[(2^2)^3\cdot 2^5]^2-2^0\right\}
    \cdot
    \left[(25^4:5^4-25^2)^{15}+(25^3)^2:(5^2)^4:5^3\right]
    \]
\end{esercizio}

\begin{soluzione}
Il primo fattore è $24^{18}/(12^{16}2^{22})-1=9-1=8$.
Il secondo è $0^{15}+5^{12-8-3}=5$. Il prodotto vale $40$.
\end{soluzione}
